Architecture

flowchart LR
  IN["Document images
photo · scan · PDF"] --> DET{"Doc-type detection
keywords, then vision LLM
(manual override)"}
  DET --> VLM["VLM direct
Groq vision model"]
  DET --> OCR["OCR + LLM
Tesseract, then text model"]
  VLM --> F["Field<T> per value
value · confidence · evidence · null_reason"]
  OCR --> F
  F --> V["Per-document validators
RxNorm · GSTIN · arithmetic · dates"]
  F --> X["Cross-document checks
rules first, LLM only for fuzzy matches"]
  V --> D{"Decision policy"}
  X --> D
  D --> A1["approve"]
  D --> A2["needs review"]
  D --> A3["flag"]
  A2 --> H["Human reviewer
makes the final call"]
  A3 --> H
  P[("Packs
claims · payables · expenses")] -. "schemas · validators · checks · policy" .-> V
  P -.-> X
  P -.-> D
